cmake_minimum_required(VERSION 3.22.1)
project(neko_player C)
set(CMAKE_C_STANDARD 99)
set(NATIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/native")
set(FFMPEG_HEADERS "${NATIVE_ROOT}/external/ffmpeg/include/${ANDROID_ABI}")
foreach(lib avutil swresample swscale avcodec avformat avfilter)
    add_library(${lib} SHARED IMPORTED)
    set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${NATIVE_ROOT}/libraries/ffmpeg/${ANDROID_ABI}/lib${lib}.so")
endforeach()
add_library(dav1d SHARED IMPORTED)
set_target_properties(dav1d PROPERTIES IMPORTED_LOCATION "${NATIVE_ROOT}/libraries/dav1d/${ANDROID_ABI}/libdav1d.so")
add_library(player SHARED native.c player.c player_audio.c player_demux.c player_diagnostics.c player_duration.c
    player_seek.c player_timing.c player_video_mediacodec.c player_video_software.c tempo.c util.c)
target_include_directories(player PRIVATE "${FFMPEG_HEADERS}")
target_compile_definitions(player PRIVATE PAWCHAN_HAS_ATEMPO=1 PAWCHAN_HAS_MEDIACODEC=1 PAWCHAN_FFMPEG_FLAVOR="ffmpeg8")
target_compile_options(player PRIVATE -Wall -Wextra -Wpedantic -Wno-deprecated-declarations)
target_link_options(player PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(player PRIVATE avfilter avformat avcodec swresample swscale avutil dav1d android OpenSLES log m mediandk)
