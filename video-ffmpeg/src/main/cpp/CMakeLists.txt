cmake_minimum_required(VERSION 3.22.1)
project(paw_playback LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(NATIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/native")
foreach(component avutil swresample swscale avcodec avformat)
    add_library(${component} SHARED IMPORTED)
    set_target_properties(${component} PROPERTIES IMPORTED_LOCATION
        "${NATIVE_ROOT}/libraries/ffmpeg/${ANDROID_ABI}/lib${component}.so")
endforeach()
add_library(pawplayback SHARED bindings.cpp transport.cpp decoder.cpp display.cpp audio.cpp frames.cpp duration_probe.cpp engine.cpp)
target_include_directories(pawplayback PRIVATE "${NATIVE_ROOT}/external/ffmpeg/include/${ANDROID_ABI}")
target_compile_options(pawplayback PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_options(pawplayback PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(pawplayback PRIVATE avformat avcodec swresample swscale avutil android log)
